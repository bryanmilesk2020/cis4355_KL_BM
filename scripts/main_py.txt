@app.route('/db-test')
def db_test():
    """Data Tier Connectivity Proof: Verifies Write/Read via Private IP"""
    try:
        with engine.connect() as conn:
            # 1. Create a verification table if it doesn't exist
            conn.execute(text("""
                CREATE TABLE IF NOT EXISTS infra_verification (
                    id INT AUTO_INCREMENT PRIMARY KEY, 
                    log_msg VARCHAR(255), 
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
                )
            """))
            
            # 2. Insert a row to prove WRITE access through the firewall
            conn.execute(text("INSERT INTO infra_verification (log_msg) VALUES ('Private Network Path Verified')"))
            conn.commit()
            
            # 3. Read it back to prove READ access through the firewall
            result = conn.execute(text("SELECT * FROM infra_verification ORDER BY id DESC LIMIT 1")).fetchone()
            
            return jsonify({
                "status": "Success",
                "connectivity": "Private Subnet -> Cloud SQL Bridge Verified",
                "database_response": {
                    "id": result[0],
                    "message": result[1],
                    "timestamp": str(result[2])
                }
            })
    except Exception as e:
        return jsonify({
            "status": "Infrastructure Error",
            "detail": str(e)
        }), 500
