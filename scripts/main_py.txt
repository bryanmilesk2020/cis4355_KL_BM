#Gives VM instance ability to function as a web server, talk to Secret Manager, and connect to the database
from google.cloud import secretmanager
from flask import Flask, jsonify
import sqlalchemy
from sqlalchemy import text

app = Flask(__name__)

#--- SECRET MANAGER INTEGRATION ---
def get_secret(secret_id):
        #Fetch a secret value from GCP secret manager at runtime
        client = secretmanager.SecretManagerServiceClient()
        name = f"projects/cis4355-bm-kl/secrets/{secret_id}/versions/latest"
        response = client.access_secret_version(request={"name": name})
        return response.payload.data.decode("UTF-8")

# --- DATABASE CREDENTIALS ---
# fetched at runtime from Secret Manager
DB_HOST = "10.10.1.3" 
DB_USER = get_secret("db-user")
DB_PASS = get_secret("db-pass")
DB_NAME = get_secret("db-name")

# Setup the database engine using the PyMySQL driver
# pool_pre_ping ensures the infrastructure connection is alive before every request
engine = sqlalchemy.create_engine(
    f"mysql+pymysql://{DB_USER}:{DB_PASS}@{DB_HOST}/{DB_NAME}",
    pool_pre_ping=True
)

#whenever user vistis / the following JSON response follows, verifying connectivity with Flask aka Application tier
@app.route('/')
def home():
    """Application Tier Identity Proof"""
    return jsonify({
        "status": "Online",
        "tier": "Application Tier (Backend VM)",
        "message": "Infrastructure Reachable"
    })

#whenever user visits /db-test in URL it returns the following JSON response and also adds that entry to the database instance
#proves that the load balancer aka static tier, web server aka application tier, and database instance aka data tier can all connect
@app.route('/db-test')
def db_test():
    """Data Tier Connectivity Proof: Verifies Write/Read via Private IP"""
    try:
        with engine.connect() as conn:
            # 1. Create a verification table if it doesn't exist
            conn.execute(text("""
                CREATE TABLE IF NOT EXISTS infra_verification (
                    id INT AUTO_INCREMENT PRIMARY KEY, 
                    log_msg VARCHAR(255), 
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
                )
            """))
            
            # 2. Insert a row to prove WRITE access through the firewall
            conn.execute(text("INSERT INTO infra_verification (log_msg) VALUES ('Private Network Path Verified')"))
            conn.commit()
            
            # 3. Read it back to prove READ access through the firewall
            result = conn.execute(text("SELECT * FROM infra_verification ORDER BY id DESC LIMIT 1")).fetchone()
            
            return jsonify({
                "status": "Success",
                "connectivity": "Private Subnet -> Cloud SQL Bridge Verified",
                "database_response": {
                    "id": result[0],
                    "message": result[1],
                    "timestamp": str(result[2])
                }
            })
    except Exception as e:
        return jsonify({
            "status": "Infrastructure Error",
            "detail": str(e)
        }), 500

if __name__ == '__main__':
    # Listen on all internal interfaces (0.0.0.0) so the Frontend/Load Balancer can find it
    app.run(host='0.0.0.0', port=5000)
